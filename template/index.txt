window._loadedScriptPromises = window._loadedScriptPromises || {};
window._loadedScriptDone = window._loadedScriptDone || {};

window.isSearchBot = function() {
  return /Googlebot|Google-InspectionTool|AdsBot-Google|Storebot-Google|bingbot/i.test(navigator.userAgent || '');
};

window.loadScriptOnce = function(src) {
  if (!src) return Promise.resolve(false);
  if (window._loadedScriptDone[src]) return Promise.resolve(true);
  if (window._loadedScriptPromises[src]) return window._loadedScriptPromises[src];

  // 既に同srcのscriptタグが存在し完了している場合
  const existing = document.querySelector(`script[src="${src}"]`);
  if (existing && existing.dataset.loaded === 'true') {
    window._loadedScriptDone[src] = true;
    return Promise.resolve(true);
  }

  window._loadedScriptPromises[src] = new Promise(resolve => {
    const script = document.createElement('script');
    script.src = src;
    script.async = false;
    script.onload = () => {
      script.dataset.loaded = 'true';
      window._loadedScriptDone[src] = true;
      resolve(true);
    };
    script.onerror = () => {
      delete window._loadedScriptPromises[src];
      script.remove();
      resolve(false);
    };
    document.head.appendChild(script);
  });

  return window._loadedScriptPromises[src];

window.detectRouteTypeFromUrl = function() {
  const params = new URLSearchParams(window.location.search);
  const hasPref = !!(window.currentPref || params.get('pref'));
  const hasSpot = !!(window.currentSpotId || params.get('spot'));
  const hasArea = !!(window.currentAreaId || params.get('area'));
  if (!hasPref) return 'region';
  if (hasSpot) return 'spot';
  if (hasArea) return 'area';
  return 'pref';
};

window.ensureRouteScripts = async function(routeType) {
  const type = routeType || window.detectRouteTypeFromUrl();
  let tasks = [];
  if (type === 'region') {
    tasks = [window.ensureRegionLoader()];
  } else if (type === 'pref' || type === 'area') {
    // PREF / AREA画面では最初から areaLoader.js も必須ロードに含める
    tasks = [
      window.ensureRegionLoader(),
      window.ensurePrefLoader(),
      window.ensureAreaLoader()
    ];
  } else if (type === 'spot') {
    tasks = [
      window.ensureRegionLoader(),
      window.ensureSpotLoader()
    ];
  } else {
    tasks = [window.ensureRegionLoader()];
  }

  await Promise.all(tasks);

  // 必須JSがWRS等で読み込み失敗して未定義のままの場合のみ安全フォールバックを補完
  if (typeof window.registerWrsFallbacks === 'function') {
    window.registerWrsFallbacks();
  }
  return true;
};

window._deferredBgScriptsRoute = null;
window._userRecoveryHooked = false;

window.triggerDeferredUserRecovery = function() {
  // 検索ボット向け省通信ルートで後回しにした処理（裏JS・道路・等深線）をユーザー操作時に後から読み込み直す
  if (window._deferredBgScriptsRoute) {
    const deferredRoute = window._deferredBgScriptsRoute;
    window._deferredBgScriptsRoute = null;
    window.preloadRemainingScripts(deferredRoute, { force: true });
  }
  if (window._depthDeferredForBot && typeof DepthMapController !== 'undefined' && window.map) {
    window._depthDeferredForBot = false;
    DepthMapController.initAndShow(window.map, { force: true });
  }
  if (window._roadDeferredForBot && !window.isSpotScreen() && typeof window.showRoadLayers === 'function') {
    window._roadDeferredForBot = false;
    window.showRoadLayers();
  }
};

function bindUserRecoveryListeners() {
  if (window._userRecoveryHooked) return;
  window._userRecoveryHooked = true;
  const onFirstInteract = () => {
    window.removeEventListener('pointerdown', onFirstInteract, true);
    window.removeEventListener('touchstart', onFirstInteract, true);
    window.removeEventListener('keydown', onFirstInteract, true);
    window.triggerDeferredUserRecovery();
  };
  window.addEventListener('pointerdown', onFirstInteract, { capture: true, passive: true });
  window.addEventListener('touchstart', onFirstInteract, { capture: true, passive: true });
  window.addEventListener('keydown', onFirstInteract, { capture: true, passive: true });
}

window.preloadRemainingScripts = function(routeType, options = {}) {
  const force = !!(options && options.force);
  const type = routeType || window.detectRouteTypeFromUrl();

  // 検索ボットには初回画面遷移用の裏JSロードをスキップして通信枠を温存し、ユーザー操作時に後から読み込み直す
  if (!force && window.isSearchBot()) {
    window._deferredBgScriptsRoute = type;
    bindUserRecoveryListeners();
    return Promise.resolve();
  }
  if (window._bgScriptsPreloadPromise) return window._bgScriptsPreloadPromise;

  const type = routeType || window.detectRouteTypeFromUrl();

  window._bgScriptsPreloadPromise = (async () => {
    try {
      if (type === 'region') {
        await window.ensurePrefLoader();
        await window.ensureAreaLoader();
        // REGION画面に滞在中であれば、areaLoader.js読み込み完了と同時に道路レイヤーを描画
        if (!window.currentPref && !window.isSpotScreen() && typeof window.showRoadLayers === 'function') {
          window.showRoadLayers();
        }
        await window.ensureSpotLoader();
      } else if (type === 'pref' || type === 'area') {
        await window.ensureSpotLoader();
      } else if (type === 'spot') {
        await window.ensureAreaLoader();
        // SPOT画面に滞在中であれば、最寄り釣具店データを取得してアクセス枠へ反映
        if (window.currentPref && window.prefShopUrl && window.markerControl && typeof window.markerControl.preloadShop01 === 'function') {
          window.markerControl.preloadShop01(window.prefShopUrl).then(() => {
            if (window.currentSpotId && window.spotData && typeof renderShopSection === 'function') {
              const activeSpot = (window.currentAreaId
                ? window.spotData.find(s => String(s.areaId) === String(window.currentAreaId) && String(s.individualId) === String(window.currentSpotId))
                : null)
                || window.spotData.find(s => String(s.individualId) === String(window.currentSpotId));
              if (activeSpot) renderShopSection(activeSpot);
            }
          });
        }
        await window.ensurePrefLoader();
        // SPOT画面に滞在中であれば、JMA警報バーを補完更新
        if (window.currentPref && window.prefData && typeof getAlertText === 'function') {
          getAlertText(window.prefData);
        }
      }
    } catch (err) {
      console.warn('バックグラウンドJS読み込みスキップ:', err);
    } finally {
      window._bgScriptsPreloadPromise = null;
    }
  })();

  return window._bgScriptsPreloadPromise;
};

function resolvePrefCodeFast(rawPref) {
  if (!rawPref) return '';
  const trimmed = String(rawPref).trim();
  if (PREF_NAME_TO_CODE[trimmed]) return PREF_NAME_TO_CODE[trimmed];
  const upper = trimmed.toUpperCase();
  if (PREF_NAME_TO_CODE[upper]) return PREF_NAME_TO_CODE[upper];
  for (const [k, code] of Object.entries(PREF_NAME_TO_CODE)) {
    if (trimmed.includes(k) || k.includes(trimmed)) return code;
  }
  return '';
}

window._rankingCacheData = window._rankingCacheData || null;
window._rankingCachePromise = window._rankingCachePromise || null;

function loadRankingCache() {
    if (window._rankingCacheData) return Promise.resolve(window._rankingCacheData);
    if (window._rankingCachePromise) return window._rankingCachePromise;

    window._rankingCachePromise = fetch('/ai-data/ranking_cache.json')
        .then(res => {
            if (!res.ok) throw new Error('ranking_cache.json HTTP ' + res.status);
            return res.json();
        })
        .then(data => {
            if (data && typeof data === 'object') {
                window._rankingCacheData = data;
                return data;
            }
            return null;
        })
        .catch(() => {
            window._rankingCachePromise = null;
            return null;
        });

    return window._rankingCachePromise;
}

// ★ 明日のオススメ釣り場ランキング差し替え関数（事前生成キャッシュ専用）
// ==========================================
async function renderSpotRanking() {
    const container = document.getElementById('spot-ranking-container');
    const section = document.getElementById('fishing-recommend-section');
    if (!container || !section) return;

    const targetPrefCode = window.currentPref ? String(window.currentPref).trim().toUpperCase() : null;
    const titleEl = section.querySelector('h2');

    // 関東トップ(REGION)で既にSSG埋め込み済みHTMLが表示されている場合は通信せず維持
    if (!targetPrefCode) {
        const embeddedList = container.querySelector('ul[data-ssg-ranking="REGION"]');
        if (embeddedList) {
            section.style.display = 'block';
            return;
        }
    }

    // 事前生成された /ai-data/ranking_cache.json から県別／関東版へ即時差し替え
    const cache = await loadRankingCache();
    if (!cache) return;

    // 非同期fetch中に別画面へ遷移していた場合は上書きしない
    const latestPrefCode = window.currentPref ? String(window.currentPref).trim().toUpperCase() : null;
    if (latestPrefCode !== targetPrefCode) return;

    const entry = targetPrefCode
        ? (cache.prefs && cache.prefs[targetPrefCode])
        : cache.region;

    if (!entry || !entry.html) return;

    section.style.display = 'block';
    if (titleEl && entry.title) {
        titleEl.textContent = entry.title;
    }
    container.innerHTML = entry.html;

    const basisTitleEl = document.getElementById('rankingBasisModalTitle');
    const basisBodyEl = document.getElementById('rankingBasisBody');
    if (basisTitleEl && entry.basisTitle) {
        basisTitleEl.textContent = entry.basisTitle;
    }
    if (basisBodyEl && entry.basisHtml) {
        basisBodyEl.innerHTML = entry.basisHtml;
    }
}

let _titleUpdateTimer = null; 

function getRegionDisplayName(regionCode = window.currentRegion) {
    const code = String(regionCode || 'KANTO').trim().toUpperCase();
    const rawName = window.regionData && window.regionData.name ? String(window.regionData.name).trim() : '';
    if (rawName && rawName.toUpperCase() !== code) return rawName;
    if (typeof REGION_MAP === 'object' && REGION_MAP !== null) {
        const mapped = Object.keys(REGION_MAP).find(name => REGION_MAP[name] === code && name !== code);
        if (mapped) return mapped;
    }
    return '関東地方';
}

function setIdealQuery(type, value) {
    const url = new URL(location.href);

    if (value) {
        // トップ画面がクエリなし(/)のままでも、pref/area/spot遷移時にregionを自動補完してURL体系を維持する
        if ((type === 'pref' || type === 'area' || type === 'spot') && !url.searchParams.get('region')) {
            url.searchParams.set('region', getRegionDisplayName());
        }
        url.searchParams.set(type, value);
    } else {
        url.searchParams.delete(type);
    }

    // クエリパラメータの並び順を region -> pref -> area -> spot に統一（その他パラメータは末尾に保持）
    const orderedKeys = ['region', 'pref', 'area', 'spot'];
    const entries = [];
    orderedKeys.forEach(k => {
        if (url.searchParams.has(k)) {
            entries.push([k, url.searchParams.get(k)]);
        }
    });
    for (const [k, v] of url.searchParams.entries()) {
        if (!orderedKeys.includes(k)) {
            entries.push([k, v]);
        }
    }
    const nextSearch = new URLSearchParams(entries).toString();
    url.search = nextSearch ? ('?' + nextSearch) : '';

    if (url.href !== location.href) {
        window.history.replaceState(null, '', url);
    }

    clearTimeout(_titleUpdateTimer);
    // タイトル更新関数を新しい名前に変更
    _titleUpdateTimer = setTimeout(updateSeoMeta, 50);
}

// ★ 新規追加：内部IDから日本語名を逆引きしてURLクエリにセットする関数
// ==========================================
async function updateQueryFromIds(pref, currentAreaId, currentSpotId) {
    
    // --------------------------------------------------
    // パターン1：第一引数 (pref) のみ存在する場合
    // --------------------------------------------------
    if (pref && !currentAreaId && !currentSpotId) {
        
        // ★ グローバル化されたURLをそのまま使う！
        if (!window.regionUrl) return;
        
        try {
            const res = await fetch(window.regionUrl);
            if (res.ok) {
                const text = await res.text();
                const lines = text.trim().split('\n');
                
                if (lines.length > 0) {
                    const headers = lines[0].split(',').map(h => h.trim());
                    const nameIdx = headers.findIndex(h => h.toLowerCase() === 'name');
                    const notesIdx = headers.findIndex(h => h.toLowerCase() === 'notes');
                    
                    if (nameIdx !== -1 && notesIdx !== -1) {
                        for (let i = 1; i < lines.length; i++) {
                            const cols = lines[i].split(',').map(c => c.trim());
                            if (cols[nameIdx] === pref) {
                                setIdealQuery('pref', cols[notesIdx]);
                                // ★ regionもついでにセットできるならここで！
                                // setIdealQuery('region', 取得したregion名など);
                                break;
                            }
                        }
                    }
                }
            }
        } catch (err) {
            console.error("Region CSVフェッチエラー:", err);
        }
    }

    // --------------------------------------------------
    // パターン2：第二引数 (currentAreaId) まで存在する場合
    // --------------------------------------------------
    else if (pref && currentAreaId && !currentSpotId) {
        if (window.areaData) {
            // 第二引数を "_" でスプリットした [0] を取得
            const splitAreaId = String(currentAreaId).split('_')[0]; 
            
            // window.areaDataの individualId 列と同じものを探す
            const area = window.areaData.find(a => String(a.individualId) === splitAreaId);
            
            if (area) {
                // name列を area クエリとしてセット
                setIdealQuery('area', area.name);
            }
        }
    }

    // --------------------------------------------------
    // パターン3：第三引数 (currentSpotId) まで存在する場合
    // --------------------------------------------------
    else if (pref && currentAreaId && currentSpotId) {
        if (window.spotData) {
            // areaId列が currentAreaId と同じで、さらに individualId が currentSpotId と同じ行を探す
            const spot = window.spotData.find(s => 
                String(s.areaId) === String(currentAreaId) && 
                String(s.individualId) === String(currentSpotId)
            );
            
            if (spot) {
                // name列を spot クエリとしてセット
                setIdealQuery('spot', spot.name);
            }
        }
    }
}

// ★ SEO対策用：HTML文字列を一括生成する関数（対象魚非表示版・region表示名正規化済み）
function buildSeoHtmlString(mainData, areasData, spotsData) {
    if (!mainData) return '';
    const regionCode = String(window.currentRegion || 'KANTO').trim().toUpperCase();
    const mappedRegionName = typeof REGION_MAP === 'object' && REGION_MAP !== null
        ? Object.keys(REGION_MAP).find(name => REGION_MAP[name] === regionCode && name !== regionCode)
        : null;
    const rawRegionName = window.regionData && window.regionData.name ? String(window.regionData.name).trim() : '';
    const regionName = (rawRegionName && rawRegionName.toUpperCase() !== regionCode)
        ? rawRegionName
        : (mappedRegionName || '関東地方');
    const prefName = mainData.notes;
    let html = '';

    areasData.forEach(area => {
        const areaKey = area.areaId + '_' + area.individualId;

        // ★ 修正：null, undefined, 空文字('') をすべて弾き、確実にzoom値があるものだけを厳選
        const areaSpots = spotsData.filter(s =>
            s.areaId === areaKey &&
            s.icon && s.icon.trim() !== '' &&
            s.zoom != null && s.zoom !== ''
        );

        if (areaSpots.length > 0) {
            html += `<h3 style="margin-top:15px; border-bottom:1px solid #ccc;">${area.name}</h3>`;
            html += `<ul style="list-style-type:none; padding-left:10px;">`;

            areaSpots.forEach(spot => {
                const url = `/?region=${encodeURIComponent(regionName)}&pref=${encodeURIComponent(prefName)}&area=${encodeURIComponent(area.name)}&spot=${encodeURIComponent(spot.name)}`;

                html += `<li style="margin:5px 0;"><a href="${url}" style="color:#0066cc; text-decoration:underline;">${spot.name}</a></li>`;
            });

            html += `</ul>`;
        }
    });
    return html;
}
window.buildSeoHtmlString = buildSeoHtmlString;

// ★ スポット用データを金庫(allspotData)から読み込む関数

async function loadLocationJSON() {
    const pref = window.currentPref;

    function parseGrid(str) {
        if (!str) return { x: null, y: null };
        const x = str.match(/x\s*:\s*(-?\d+)/);
        const y = str.match(/y\s*:\s*(-?\d+)/);
        return {
            x: x ? parseInt(x[1]) : null,
            y: y ? parseInt(y[1]) : null
        };
    }

    // 魚データは裏側で非同期取得（タイル描画を止めない）
    if (window.fishUrl && typeof loadFishDataAsync === 'function') {
        loadFishDataAsync(pref, window.fishUrl);
    }

    // 分岐A：既に県データが生成されている場合
    if (
        window[`${pref}_prefData`] &&
        window[`${pref}_areaData`] &&
        window[`${pref}_spotData`]
    ) {
        window.prefData = window[`${pref}_prefData`];
        window.areaData = window[`${pref}_areaData`];
        window.spotData = window[`${pref}_spotData`];
        window.prefBounds = window[`${pref}_prefBounds`] || null;

        if (window.fishDataCache && window.fishDataCache[pref] && typeof applyFishDataToSpots === 'function') {
            applyFishDataToSpots(window.spotData, window.fishDataCache[pref]);
        }

        if (window[`${pref}_areaGraph`]) {
            window.areaGraph = window[`${pref}_areaGraph`];
        } else if (typeof buildAreaGraphFromGrid === 'function') {
            buildAreaGraphFromGrid(window.areaData);
            window[`${pref}_areaGraph`] = window.areaGraph;
        }

        const container = document.getElementById('seo-link-container');
        const titleSpan = document.getElementById('seo-list-title');
        if (container && titleSpan && window.prefData) {
            titleSpan.textContent = `${window.prefData.notes}の釣りスポット一覧を見る`;
            container.innerHTML = window[`${pref}_seoHtml`] || '';
        }

        return {
            main: window.prefData,
            areas: window.areaData,
            spots: window.spotData
        };
    }

    // 分岐B：金庫(allspotData)から対象県を抽出
    const vaultData = window.allspotData || window.ALL_REGION_SPOTS || [];
    let main = null;
    const areas = [];
    const spots = [];

    const targetRows = vaultData.filter(row => {
        return (
            row._prefCode === pref ||
            (!row.areaId && row.name === pref) ||
            row.areaId === pref ||
            (row.areaId && typeof row.areaId === 'string' && row.areaId.startsWith(pref + '_'))
        );
    });

    targetRows.forEach(row => {
        row.squareX = null;
        row.squareY = null;
    });

    targetRows.forEach(row => {
        if ((!row.areaId || row.individualId === 'parent') && (row.name === pref || row._prefCode === pref)) {
            main = row;
        }
    });

    targetRows.forEach(row => {
        if ((row.areaId || '').trim() === pref) {
            if (row.url && row.url.includes('x:') && row.url.includes('y:')) {
                const grid = parseGrid(row.url);
                row.squareX = grid.x;
                row.squareY = grid.y;
            }
            areas.push(row);
        }
    });

    targetRows.forEach(row => {
        if (row.type === 'pref' || row.type === 'area') return;
        const icon = row.icon;
        if (icon && icon.trim() !== '') {
            spots.push(row);
        }
    });

    if (window.fishDataCache && window.fishDataCache[pref] && typeof applyFishDataToSpots === 'function') {
        applyFishDataToSpots(spots, window.fishDataCache[pref]);
    }

    window.prefData = main;
    window.areaData = areas;
    window.spotData = spots;
    window.prefBounds = null;

    if (spots.length > 0 && typeof L !== 'undefined') {
        let minLat = Infinity, maxLat = -Infinity;
        let minLng = Infinity, maxLng = -Infinity;
        spots.forEach(spot => {
            const lat = Number(spot.lat);
            const lng = Number(spot.lng);
            if (Number.isFinite(lat) && Number.isFinite(lng)) {
                minLat = Math.min(minLat, lat);
                maxLat = Math.max(maxLat, lat);
                minLng = Math.min(minLng, lng);
                maxLng = Math.max(maxLng, lng);
            }
        });
        const latBuffer = Math.max((maxLat - minLat) * 0.1, 0.05);
        const lngBuffer = Math.max((maxLng - minLng) * 0.1, 0.05);
        window.prefBounds = L.latLngBounds(
            [minLat - latBuffer, minLng - lngBuffer],
            [maxLat + latBuffer, maxLng + lngBuffer]
        );
    }

    window[`${pref}_prefData`] = main;
    window[`${pref}_areaData`] = areas;
    window[`${pref}_spotData`] = spots;
    window[`${pref}_prefBounds`] = window.prefBounds;

    if (typeof buildAreaGraphFromGrid === 'function') {
        buildAreaGraphFromGrid(areas);
        window[`${pref}_areaGraph`] = window.areaGraph;
    }

    if (typeof buildSeoHtmlString === 'function') {
        const seoHtml = buildSeoHtmlString(main, areas, spots);
        window[`${pref}_seoHtml`] = seoHtml;

        const container = document.getElementById('seo-link-container');
        const titleSpan = document.getElementById('seo-list-title');
        if (container && titleSpan && main) {
            titleSpan.textContent = `${main.notes}の釣りスポット一覧を見る`;
            container.innerHTML = seoHtml;
        }
    }

    return { main, areas, spots };
}
async function drawLocation(name, lat, lng, zoom, options = {}) {
  if (typeof L === 'undefined') {
    if (typeof window._flushTileQueue === 'function') {
      window._flushTileQueue();
    }
    return;
  }

  const defaultOptions = {
    center: [lat, lng],
    zoom,
    zoomControl: false,
    scrollWheelZoom: false,
    dragging: false,
    doubleClickZoom: false,
    boxZoom: false,
    keyboard: false,
    touchZoom: false,
    fadeAnimation: false // ★初期タイルの0.2秒フェード待ちによるLCP遅延を防止
  };

  const mapOptions = { ...defaultOptions, ...options };

  if (!window.map) {

    window.map = L.map('lf-map', mapOptions);

    window.map.options.zoomSnap = 0.5;
    window.map.options.zoomDelta = 0.5;
    window.map.attributionControl.setPosition('topright');

    window.phase1Group = L.layerGroup().addTo(window.map);
    window.phase2Group = L.layerGroup().addTo(window.map);

    window.gsiLayer = L.tileLayer(
      window.currentSpotBaseTile || window.gsiLayers.ort,
      {
        attribution: '© 国土地理院',
        keepBuffer: 0, // ★ 初回表示時は画面外の余分なタイル取得をゼロにし、表示領域のタイルだけに全帯域を集中させる
        updateWhenIdle: true
      }
    );

    // タイル画像の取得優先度と非同期デコードを指定する。
    window.gsiLayer.on('tileloadstart', (e) => {
        if (e.tile) {
            e.tile.setAttribute('fetchpriority', 'high');
            e.tile.decoding = 'async';
        }
    });

    // 初期ロード時は、ベースタイルの読み込みと画面反映が終わるまで呼び出し元を待機させる。
    let resolveInitialTileLoad;
    const initialTileLoadPromise = window.isInitialMapLoad
        ? new Promise(resolve => { resolveInitialTileLoad = resolve; })
        : null;

    // ベースタイルの読み込み完了イベントで keepBuffer を 2 / updateWhenIdle を false / fadeAnimation を true に戻し、後続キューを順次スタート
    window.gsiLayer.once('load', () => {
        window.gsiLayer.options.keepBuffer = 2;
        window.gsiLayer.options.updateWhenIdle = false;
        if (window.map && window.map.options) {
            window.map.options.fadeAnimation = true;
        }
        requestAnimationFrame(() => {
            if (typeof window._flushTileQueue === 'function') {
                window._flushTileQueue();
            }
            if (resolveInitialTileLoad) {
                requestAnimationFrame(resolveInitialTileLoad);
            }
        });
    });

    window.gsiLayer.addTo(window.map);

    // ★ 等深線レイヤー(5ファイル)は、ベースタイル描画完了後のキューで開始
    if (typeof DepthMapController !== 'undefined') {
        window.runAfterTileLoad(() => {
            return DepthMapController.initAndShow(window.map);
        });
    }

    if (initialTileLoadPromise) await initialTileLoadPromise;
    return;
  }

  // 初期URL復帰中の地図移動も、移動完了後の表示タイルが揃うまで待つ。
  let initialMoveTilePromise = null;
  if (window.isInitialMapLoad && window.gsiLayer) {
      const currentCenter = window.map.getCenter();
      const currentZoom = window.map.getZoom();
      const isSameView =
          Math.abs(currentCenter.lat - lat) < 0.000001 &&
          Math.abs(currentCenter.lng - lng) < 0.000001 &&
          currentZoom === zoom;

      if (isSameView) {
          initialMoveTilePromise = Promise.resolve();
      } else {
          initialMoveTilePromise = new Promise(resolve => {
              window.map.once('moveend', () => {
                  const finishAfterPaint = () => {
                      requestAnimationFrame(() => requestAnimationFrame(resolve));
                  };
                  // GridLayer自身のmoveend処理が終わってから、最終表示範囲のタイル状態を確認する。
                  requestAnimationFrame(() => {
                      if (window.gsiLayer._loading) {
                          window.gsiLayer.once('load', finishAfterPaint);
                      } else {
                          finishAfterPaint();
                      }
                  });
              });
          });
      }
  }

  window.map.flyTo([lat, lng], zoom, { duration: 0.5 });

  mapOptions.scrollWheelZoom
    ? window.map.scrollWheelZoom.enable()
    : window.map.scrollWheelZoom.disable();

  mapOptions.dragging
    ? window.map.dragging.enable()
    : window.map.dragging.disable();

  mapOptions.doubleClickZoom
    ? window.map.doubleClickZoom.enable()
    : window.map.doubleClickZoom.disable();

  mapOptions.boxZoom
    ? window.map.boxZoom.enable()
    : window.map.boxZoom.disable();

  mapOptions.keyboard
    ? window.map.keyboard.enable()
    : window.map.keyboard.disable();

  mapOptions.touchZoom
    ? window.map.touchZoom.enable()
    : window.map.touchZoom.disable();

  if (initialMoveTilePromise) await initialMoveTilePromise;
}

// ★ ユーザー操作待ちを完全廃止：ベースタイルの load 完了後、1フレームに1タスクずつ順番にバトンリレー実行する
window._tileLoadCallbacks = [];
window._isFlushingTileQueue = false;

// ★ 1フレームの描画を確実にブラウザへ返してから次の処理へ進むフレーム待機ヘルパー
window._yieldFrame = function() {
    return new Promise(resolve => {
        requestAnimationFrame(() => {
            const ch = new MessageChannel();
            ch.port1.onmessage = () => resolve();
            ch.port2.postMessage(null);
        });
    });
};

window._flushTileQueue = function() {
    window._firstTileLoaded = true;
    if (window._isFlushingTileQueue) return;
    window._isFlushingTileQueue = true;

    const step = async () => {
        if (window._tileLoadCallbacks.length === 0) {
            window._isFlushingTileQueue = false;
            return;
        }
        const fn = window._tileLoadCallbacks.shift();
        try {
            await fn();
        } catch (e) {
            console.error(e);
        }
        await window._yieldFrame();
        step();
    };

    window._yieldFrame().then(step);
};

window.runAfterTileLoad = function(callback) {
    window._tileLoadCallbacks.push(callback);
    if (window._firstTileLoaded) {
        window._flushTileQueue();
    }
};

// 遷移開始時に戻るボタンとピンチインを確実にロックして隠す（zoomToSpot と同設計）
function lockBackNavigation(options = {}) {
    window.goBackGuard = true;
    const gen = (window._navUnlockGen = (window._navUnlockGen || 0) + 1);
    const hideBtn = options.hideButton !== false;
    const backBtn = document.getElementById('map-back-btn');
    if (backBtn && hideBtn) {

        backBtn.style.transition = 'opacity 0.25s ease';
        backBtn.style.opacity = '0';
    }

}

// タイル読み込みPromise（読み込み済み・タイムアウト時は即解決）
function waitTileLayerReady(layer, timeoutMs = 1200) {
    return new Promise(resolve => {
        if (!layer) { resolve(); return; }
        const isLoading = !!(
            (typeof layer.isLoading === 'function' && layer.isLoading()) ||
            layer._loading
        );
        if (!isLoading) {
            requestAnimationFrame(() => resolve());
            return;
        }
        let done = false;
        const finish = () => {
            if (done) return;
            done = true;
            clearTimeout(timer);
            layer.off('load', finish);
            requestAnimationFrame(() => resolve());
        };
        layer.once('load', finish);
        const timer = setTimeout(finish, timeoutMs);
    });
}

// 画面遷移の移動・タイル・フェード・必須処理がすべて完了してから戻る操作を解禁する
function unlockBackNavigationWhenReady(options = {}) {
    const gen = options.gen != null ? options.gen : (window._navUnlockGen = (window._navUnlockGen || 0) + 1);
    const showButton = options.showButton !== false;
    const waitPromises = Array.isArray(options.promises) ? options.promises.filter(Boolean) : [];
    const maxWaitMs = Number(options.maxWaitMs) || 3000;

    const finishUnlock = () => {
        if (window._navUnlockGen !== gen) return;
        clearTimeout(safetyTimer);
        const backBtn = document.getElementById('map-back-btn');
        if (backBtn) {
            if (showButton) {
                backBtn.style.display = 'block';
                backBtn.style.pointerEvents = 'auto';
                requestAnimationFrame(() => {
                    if (window._navUnlockGen !== gen) return;
                    backBtn.style.transition = 'opacity 0.4s ease';
                    backBtn.style.opacity = '1';
                });
            } else {
                backBtn.style.display = 'none';
                backBtn.style.pointerEvents = 'auto';
            }
        }
        setTimeout(() => {
            if (window._navUnlockGen === gen) {
                window.goBackGuard = false;
            }
        }, 150);
    };

    const safetyTimer = setTimeout(finishUnlock, maxWaitMs);
    Promise.allSettled(waitPromises).then(() => {
        requestAnimationFrame(() => requestAnimationFrame(finishUnlock));
    });
}

        if (typeof enablePrefSwipe === 'function') enablePrefSwipe();

        const waitTasks = [waitTileLayerReady(window.gsiLayer, 1000)];

    (async () => {
        try {
            if (typeof pref_management !== 'function' && typeof window.ensurePrefLoader === 'function') {
                await window.ensurePrefLoader();
            }
            let drawn = false;
            try {
                if (typeof pref_management === 'function') {
                    drawn = await pref_management(window.currentPref);
                }
            } catch (err) {
                console.error('pref_management エラー（従来の描画で続行）:', err);
            }

            // pref_management が使えなかった場合のみ、従来どおりここで描画する
            if (!drawn) {
                drawLocation(window.prefData.name, window.prefData.lat, window.prefData.lng, window.prefData.zoom);

                const centerPref = window.map.getCenter();
                const isSamePref = Math.abs(centerPref.lat - window.prefData.lat) < 0.0001 &&
                                   Math.abs(centerPref.lng - window.prefData.lng) < 0.0001 &&
                                   window.map.getZoom() === window.prefData.zoom;
                if (!isSamePref) {
                    await new Promise(resolve => {
                        const timer = setTimeout(resolve, 1500); // moveend が来ない場合の保険
                        window.map.once('moveend', () => { clearTimeout(timer); resolve(); });
                    });
                }
            }

            completePrefReturn();
        } catch (err) {
            console.error('エリア→県の復帰エラー:', err);
            unlockBackNavigationWhenReady({ gen: navGen, showButton: true, maxWaitMs: 300 });
        }
    })();
}

if (typeof window.toWeatherIcon !== 'function') {

if (typeof window.applyFishDataToSpots !== 'function') {
    window.applyFishDataToSpots = function(spots, fishData) {
        if (!Array.isArray(spots)) return;
        if (!fishData || typeof fishData !== 'object') {
            spots.forEach(spot => { spot.URL = ""; });
            return;
        }
        const fishDict = {};
        for (const regKey in fishData) {
            const areaSpots = fishData[regKey];
            if (!areaSpots || typeof areaSpots !== 'object') continue;
            for (const spotName in areaSpots) {
                fishDict[spotName] = areaSpots[spotName];
            }
        }
        spots.forEach(spot => {
            const spotFishData = fishDict[spot.name];
            if (!spotFishData) { spot.URL = ""; return; }
            const fishList = [];
            for (const fishName in spotFishData) {
                const info = spotFishData[fishName];
                if (info && typeof info.coords === 'string' && info.coords !== '') {
                    info.coords.split('|').forEach(pt => {
                        const [lat, lng] = pt.split(',');
                        if (lat && lng) fishList.push(`${fishName}|${lat}|${lng}`);
                    });
                }
            }
            spot.URL = fishList.join(',');
        });
    };
}

if (typeof window.loadFishDataForPref !== 'function') {
    window.loadFishDataForPref = function(pref, fishUrl) {
        if (!pref || !fishUrl) return Promise.resolve(null);
        if (window.fishDataCache[pref]) {
            const cachedSpots = window.prefSpotData?.[pref];
            if (Array.isArray(cachedSpots)) window.applyFishDataToSpots(cachedSpots, window.fishDataCache[pref]);
            return Promise.resolve(window.fishDataCache[pref]);
        }
        if (window.fishDataPromises[pref]) return window.fishDataPromises[pref];
        window.fishDataPromises[pref] = fetch(fishUrl)
            .then(res => {

                if (fishData && typeof fishData === 'object') {
                    window.fishDataCache[pref] = fishData;
                    const targetSpots = window.prefSpotData?.[pref];
                    if (Array.isArray(targetSpots)) window.applyFishDataToSpots(targetSpots, fishData);

            })
            .catch(() => {

if (typeof window.loadRegionCsvData !== 'function') {
    window.loadRegionCsvData = function() {
        const regionCode = window.currentRegion || 'KANTO';
        const rUrl = window.regionUrl || `/${regionCode}/${regionCode}_region.csv`;
        if (window.regionCsvParsedData && window.regionCsvParsedRegion === regionCode && window.regionCsvParsedUrl === rUrl) {
            return Promise.resolve(window.regionCsvParsedData);
        }
        if (window.regionCsvParsedPromise && window.regionCsvParsedRegion === regionCode && window.regionCsvParsedUrl === rUrl) {
            return window.regionCsvParsedPromise;
        }
        window.regionCsvParsedRegion = regionCode;
        window.regionCsvParsedUrl = rUrl;
        window.regionCsvParsedPromise = fetch(rUrl)
            .then(res => {
                if (!res.ok) throw new Error('Region CSV fetch failed: ' + res.status);
                return res.text();
            })
            .then(text => {
                const lines = text.trim().split('\n');
                const headers = lines[0].split(',').map(h => h.trim().toLowerCase());
                const rows = [];
                for (let i = 1; i < lines.length; i++) {
                    if (!lines[i]) continue;
                    const cols = lines[i].split(',').map(c => c.trim());
                    const row = {};
                    headers.forEach((h, idx) => { row[h] = cols[idx] || ''; });
                    rows.push(row);
                }
                const prefRows = [];
                for (const row of rows) {
                    if (row.type !== 'pref' || !row.url) continue;
                    const parts = row.url.split('/');
                    let prefCode = '', lat = NaN, lng = NaN;
                    if (parts.length >= 4) {
                        prefCode = parts[1].split('_')[0];
                        lat = parseFloat(parts[2]);
                        lng = parseFloat(parts[3]);
                    } else if (parts.length === 3) {
                        prefCode = parts[0].split('_')[0];
                        lat = parseFloat(parts[1]);
                        lng = parseFloat(parts[2]);
                    }
                    if (prefCode) prefRows.push({ prefCode, prefName: row.notes || '', lat, lng, row });
                }
                const parsedData = { regionCode, url: rUrl, headers, rows, prefRows };
                window.regionCsvParsedData = parsedData;
                return parsedData;
            });
        return window.regionCsvParsedPromise;
    };
}

if (typeof window.loadPrefectureSpots !== 'function') {
    window.loadPrefectureSpots = async function(prefCode, mode = 'select') {
        const regionCode = window.currentRegion || 'KANTO';
        const fetchPref = (code, priority = 'low') => {
            if (!code) return Promise.resolve([]);
            if (Array.isArray(window.prefSpotData[code])) return Promise.resolve(window.prefSpotData[code]);
            if (window.prefSpotPromises[code]) return window.prefSpotPromises[code];
            const url = `/${regionCode}/${code}_location.json`;
            const maxRetries = priority === 'high' ? 1 : 0;
            const doFetch = attempt => fetch(url, { priority })
                .then(r => {
                    if (!r.ok) throw new Error('HTTP ' + r.status);
                    return r.json();
                })
                .then(data => {
                    if (!Array.isArray(data)) return [];
                    data.forEach(s => { s._prefCode = code; });
                    window.prefSpotData[code] = data;
                    window._loadedPrefSet.add(code);
                    return data;
                })
                .catch(() => {
                    if (attempt < maxRetries) {
                        return new Promise(resolve => setTimeout(resolve, 300)).then(() => doFetch(attempt + 1));
                    }
                    delete window.prefSpotPromises[code];
                    return [];
                });
            window.prefSpotPromises[code] = doFetch(0);
            return window.prefSpotPromises[code];
        };
        if (mode === 'select' && prefCode) {
            await fetchPref(prefCode, 'high');
            window.allspotData = Object.values(window.prefSpotData).flat();
            window.ALL_REGION_SPOTS = window.allspotData;
            return window.prefSpotData[prefCode] || [];
        }
        try {
            const csvData = await window.loadRegionCsvData();
            const prefList = [...new Set((csvData.prefRows || []).map(r => r.prefCode).filter(Boolean))];
            const targets = mode === 'other' ? prefList.filter(c => c !== prefCode) : prefList;
            await Promise.all(targets.map(c => fetchPref(c, 'low')));
        } catch (e) {}
        window.allspotData = Object.values(window.prefSpotData).flat();
        window.ALL_REGION_SPOTS = window.allspotData;
        return window.allspotData;
    };
}

            .then(data => {
                window.prefDictionaryCache[key] = data;

                return data;

            .catch(() => {
                if (attempt < 2) {
                    return new Promise(resolve => setTimeout(resolve, 250 * (attempt + 1))).then(() => doFetch(attempt + 1));
                }
                delete window.prefDictionaryPromises[key];
                return null;
            });
        window.prefDictionaryPromises[key] = doFetch(0);
        return window.prefDictionaryPromises[key];

// ★ 4. URLクエリから直接システム変数をセットする関数
// ==========================================
async function applyQueryToStateFromCsv() {
    const params = new URLSearchParams(window.location.search);
    const qPref = params.get('pref');
    const qArea = params.get('area');
    const qSpot = params.get('spot');

    if (!qPref) return;

    // 1. 静的マップで県コードを即解決（KANTO_region.csv の通信待ちをバイパス）
    const fastCode = resolvePrefCodeFast(qPref);
    if (fastCode) {
        window.currentPref = fastCode;
        window.prefLocationUrl = `/${window.currentRegion}/${window.currentPref}_location.json`;
        // SPOT画面なら location.json と同時に逆引き辞書JSONも最優先並列発射しておく
        if (qSpot && typeof window.loadPrefDictionaryJson === 'function') {
            window.loadPrefDictionaryJson(fastCode);
        }
    } else if (window.regionUrl) {
        try {
            const csvData = await loadRegionCsvData();

            if (csvData.prefRows) {
                for (const pref of csvData.prefRows) {
                    const csvNote = pref.prefName || '';
                    const csvName = pref.row.name || '';
                    const extractedCode = pref.prefCode;

                    if (
                        csvNote && (
                            csvNote === qPref ||
                            csvNote.includes(qPref) ||
                            qPref.includes(csvNote) ||
                            csvName === qPref ||
                            extractedCode === qPref
                        )
                    ) {
                        window.currentPref = csvName || extractedCode;
                        window.prefLocationUrl = `/${window.currentRegion}/${window.currentPref}_location.json`;
                        break;
                    }
                }
            }
        } catch (err) {
            console.error("Region CSV逆引きエラー:", err);
        }
    }

    let vaultData = [];
    if (window.currentPref) {
        vaultData = await loadPrefectureSpots(window.currentPref, 'select');
    } else {
        vaultData = await loadPrefectureSpots(null, 'all');
        const parentRow = vaultData.find(r =>
            r.individualId === 'parent' &&
            r.notes && (
                r.notes === qPref ||
                r.notes.includes(qPref) ||
                qPref.includes(r.notes) ||
                r.name === qPref
            )
        );
        if (parentRow) {
            window.currentPref = parentRow.name;
            window.prefLocationUrl = `/${window.currentRegion}/${window.currentPref}_location.json`;
        }
    }

    if (!vaultData || vaultData.length === 0) return;

    if (qArea || qSpot) {
        const prefRows = window.currentPref
            ? vaultData.filter(r => r._prefCode === window.currentPref || (r.areaId && String(r.areaId).startsWith(window.currentPref)))
            : vaultData;

        for (const row of prefRows) {
            const rowName = row.name;
            if (qArea && rowName === qArea && row.areaId && row.individualId) {
                window.currentAreaId = row.areaId + "_" + row.individualId;
            }
            if (qSpot && rowName === qSpot && row.type !== 'pref' && row.type !== 'area') {
                window.currentSpotId = row.individualId;
                if (!qArea && row.areaId) {
                    window.currentAreaId = row.areaId;
                }
            }
        }
    }
}

// ★ ページ初期ロード時のメインフロー（LCPタイル描画最優先版）

document.addEventListener('DOMContentLoaded', async () => {

    const params = new URLSearchParams(window.location.search);

    // トップページ(/)ではURLへ?region=関東地方を強制付与しない（クエリ有無どちらでも同じ動作に統一）
    window.currentRegion = getRegionCode();
    window.regionUrl = `/${window.currentRegion}/${window.currentRegion}_region.csv`;

    // 到達画面に応じた必須JSのみ先に動的ロードする
    const initialRouteType = window.detectRouteTypeFromUrl();
    await window.ensureRouteScripts(initialRouteType);

    // クエリ反映前の状態を確認し、ページ初期表示かどうかを確定する。
    window.isInitialMapLoad = (
        !window.currentPref &&
        !window.currentAreaId &&
        !window.currentSpotId &&
        !window.prefData &&
        (!Array.isArray(window.areaData) || window.areaData.length === 0) &&
        (!Array.isArray(window.spotData) || window.spotData.length === 0) &&
        !window.currentHash
    );

    if (params.get('pref')) {
        await applyQueryToStateFromCsv();
    }

    if (typeof routeUpdate === 'function') {
        await routeUpdate();
    }

    // 初期URL復帰が完了したら、以後のユーザー操作ではタイル待機を行わない。
    window.isInitialMapLoad = false;

    // 初期表示完了後、通常ブラウザのみ残りの遷移用JSをバックグラウンド先読みする
    setTimeout(() => {
        if (typeof window.preloadRemainingScripts === 'function') {
            window.preloadRemainingScripts(initialRouteType);
        }
    }, 0);

    // ★ タイル描画完了後、ドット・道路・等深線の次に裏データと記事一覧を順番にバトンリレー実行する
    if (typeof window.runAfterTileLoad === 'function') {
        window.runAfterTileLoad(() => {
            if (!window.currentPref && typeof loadPrefectureSpots === 'function') return loadPrefectureSpots(null, 'all');
        });
        window.runAfterTileLoad(() => {
            if (typeof loadICData === 'function') return loadICData();
        });
        window.runAfterTileLoad(() => {
            if (typeof loadAndRenderCategorizedArticles === 'function') {
                return loadAndRenderCategorizedArticles('article-list-container');
            }
        });
    }
});

// ★ URL直打ち（初期ロード）からの復帰専用：routeUpdate関数
// （PREF画面のロード・描画は prefLoader.js の pref_management に任せる）
// ==========================================
async function routeUpdate() {
    // 旧URLなどに残る region=KANTO を表示名へ正規化する（すでに表示名またはクエリ無しの場合は書き換えない）。
    const regionParam = new URLSearchParams(window.location.search).get('region');
    const regionName = regionParam && Object.keys(REGION_MAP).find(name => REGION_MAP[name] === regionParam && name !== regionParam);
    if (regionName) setIdealQuery('region', regionName);

    if (!window.currentPref) {
        await window.ensureRouteScripts('region');
        if (typeof region_management === 'function') await region_management();
    } else {
        // ★ URLにspotクエリがあるのに初回fetch失敗等でcurrentSpotIdが未確定なら、県画面へ落ちる前に1回だけ再解決する
        const urlParams = new URLSearchParams(window.location.search);
        const qSpotName = urlParams.get('spot');
        if (qSpotName || window.currentSpotId) {
            await window.ensureRouteScripts('spot');
        } else {
            // PREF / AREA画面では最初から regionLoader + prefLoader + areaLoader を必須ロード
            await window.ensureRouteScripts(window.currentAreaId ? 'area' : 'pref');
        }

        if (qSpotName && !window.currentSpotId) {
            await applyQueryToStateFromCsv();
        }

        // ★ SPOT画面の最優先ロード（対象スポットのデータのみ先行取得 → drawLocation完了 → 後続描画＆所属県の天気・辞書先行取得）
        let spotHandled = false;
        if (window.currentSpotId && typeof spot_management === 'function') {
            try {
                spotHandled = await spot_management(window.currentPref, window.currentAreaId, window.currentSpotId);
            } catch (err) {
                console.error('spot_management エラー（従来フローで続行）:', err);
            }
        }

        // ★ URLにspotクエリがあるのに location.json 取得失敗等でスポット未確定のままなら、
        //   県画面への誤遷移を止めつつ、クエリ＋逆引き辞書だけでスポット詳細とSEOメタを完成させる
        if (qSpotName && !window.currentSpotId) {
            const qPrefName = urlParams.get('pref') || '';
            const qAreaName = urlParams.get('area') || '';
            if (typeof applyInstantSeoFromQuery === 'function') applyInstantSeoFromQuery();
            if (window.currentPref && typeof window.loadPrefDictionaryJson === 'function') {
                const dict = await window.loadPrefDictionaryJson(window.currentPref);
                const entry = dict && dict[qSpotName];
                if (entry) {
                    const fakeSpot = {
                        name: qSpotName,
                        lat: Number(entry.lat) || 0,
                        lng: Number(entry.lng) || 0,
                        _prefCode: window.currentPref
                    };
                    if (typeof renderAccessInfo === 'function' && fakeSpot.lat && fakeSpot.lng) {
                        renderAccessInfo(fakeSpot);
                    } else if (entry.description) {
                        const accessBox = document.getElementById('accessInfoBox');
                        if (accessBox) {
                            const esc = s => String(s || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
                            accessBox.innerHTML = `<div id="spot-access-main"><div class="spot-header-wrap"><div class="spot-water-regional">${esc([qPrefName, qAreaName].filter(Boolean).join(' '))}</div><h3 class="spot-main-title">${esc(qSpotName)}</h3></div><p class="spot-description">${esc(entry.description)}</p></div>`;
                            accessBox.style.display = 'block';
                        }
                    }
                    const descMeta = document.querySelector('meta[name="description"]');
                    if (descMeta && entry.description) descMeta.setAttribute('content', entry.description);
                    const ogDescMeta = document.querySelector('meta[property="og:description"]');
                    if (ogDescMeta && entry.description) ogDescMeta.setAttribute('content', entry.description);
                }
            }
            if (typeof renderSpotRanking === 'function') renderSpotRanking();
            if (typeof syncGuideSections === 'function') syncGuideSections();
            return;
        }

        if (!spotHandled) {
            // PREF / AREA画面（またはSPOTフォールバック時）は prefLoader + areaLoader を確実に揃える
            await window.ensureRouteScripts(window.currentAreaId ? 'area' : 'pref');
        }

        let prefDrawn = false;
        if (!spotHandled && typeof pref_management === 'function') {
            try {
                prefDrawn = await pref_management(window.currentPref);
            } catch (err) {
                console.error('pref_management エラー（従来フローで続行）:', err);
            }
        }

        if (!spotHandled) {
            const isReady = await checkAndLoadRouteData();
            if (!isReady) return;
        }

        if (spotHandled) {
            // spot_management で描画・先行ロード開始済み
        } else if (window.currentSpotId) {
            const spotId = String(window.currentSpotId).split("_");
            const safeAreaId = window.currentAreaId || ""; 
            
            if (typeof prepareFishForArea === 'function') await prepareFishForArea(safeAreaId);
            
            const list = window.spotData.filter(s => String(s.areaId) === safeAreaId);
            const targetIndivId = spotId[2] || spotId[0]; 
            const spot = list.find(s => String(s.individualId) === String(targetIndivId));
            
            if (spot) {
                if (window.prefData) setIdealQuery('pref', window.prefData.notes);
                const parentArea = window.areaData.find(a => safeAreaId.endsWith(a.individualId));
                
                if (parentArea) {
                    setIdealQuery('area', parentArea.name);

                    if (typeof drawLocation === 'function') {

        }

    // ★ 各分岐の共通した最後に一度、SNS釣果データを取得し、完了後にソナーマーカーを呼び出す
    if (typeof window.loadSnsCatchData === 'function') {

        if (typeof window.showSnsSonarLayer === 'function') {
            await window.showSnsSonarLayer();
        }

window.selectPrefecture = async function (csvParam, fallbackPrefName, options = {}) {
    const navGen = lockBackNavigation({ hideButton: true });

    // ★ REGION画面から裏ロード完了前に高速クリックされた場合でも、PREF必須JSを揃えてから遷移する
    if (typeof window.ensureRouteScripts === 'function') {
        await window.ensureRouteScripts('pref');
    }

    // ★ 追加：県画面に移行する際はRegion用のニアレストクリックを確実に解除する
    if (window.regionNearestClickHandler && window.map) {
        window.map.off('click', window.regionNearestClickHandler);
    }

    let region = window.currentRegion || 'KANTO';
    let prefCode = csvParam;

    if (csvParam.includes('_location')) {
        prefCode = csvParam.split('_')[0];
    } else if (csvParam.includes('_')) {
        const parts = csvParam.split('_');
        region = parts[0];
        prefCode = parts[1];
    } else if (csvParam.includes('.csv')) {
        const match =
            csvParam.match(/\/([A-Za-z-]+)_/) ||
            csvParam.match(/^([A-Za-z-]+)_/);

        if (match) {
            prefCode = match[1];
        }
    }

    window.currentRegion = region;
    window.currentPref = prefCode;
    window.currentAreaId = null;
    window.currentSpotId = null;

    // 県切り替え開始時に、前の県の取得結果が後から反映されないよう無効化
    window._jmaAlertRequestId = (window._jmaAlertRequestId || 0) + 1;
    window._currentAlertUrl = null;
    if (typeof clearMapAlert === 'function') clearMapAlert();

    window.prefBounds = null;
    window.areaBounds = null;

    window.prefLocationUrl = `/${region}/${prefCode}_location.json`;
    window.prefWeatherUrl = `/data/${prefCode}_weather_v2.json`;
    window.prefInlandWeatherUrl = `/data/${prefCode}_InLandwhether.json`;
    window.fishUrl = `/${region}/${prefCode}_fish.json`;
    window.prefShopUrl = `/${region}/${prefCode}_shop.json`;

    let prefName = fallbackPrefName;

    // =====================================================
    // 県データの抽出
    // =====================================================
    if (window.ALL_REGION_SPOTS && window.ALL_REGION_SPOTS.length > 0) {
        const targetPrefData = window.ALL_REGION_SPOTS.find(
            r => r.name === prefCode && r.individualId === "parent"
        );

        if (targetPrefData) {
            window.prefData = targetPrefData;
            prefName = targetPrefData.notes;
        }
    }

    if (prefName) {
        setIdealQuery('pref', prefName);
    } else {
        setIdealQuery('pref', prefCode);
    }

    setIdealQuery('area', null);
    setIdealQuery('spot', null);
    if (typeof syncGuideSections === 'function') syncGuideSections();

    // ★ 差し替え：drawLocation と他県JSONの並列取得は pref_management に任せる
    //   （内部で既存マップからの移動完了まで待ってから戻る）
    let drawn = false;
    if (typeof pref_management === 'function') {
        try {
            drawn = await pref_management(prefCode);
        } catch (err) {
            console.error('pref_management エラー（従来フローで続行）:', err);
        }
    }

    if (typeof loadPrefectureData === 'function') {
        await loadPrefectureData(prefCode);
    }

    if (window.prefData && window.map) {
        const targetZoom = window.prefData.zoom;

        // pref_management が使えなかった場合のみ、従来どおりここで描画する
        if (!drawn) {
            await loadLocationJSON();

            drawLocation(
                window.prefData.name,
                window.prefData.lat,
                window.prefData.lng,
                targetZoom
            );
        }

        const finalizePref = () => {
            window.map.invalidateSize(false);

            if (
                typeof loadPrefectureData === 'function' &&
                (!window.spotData || window.spotData.length === 0)
            ) {
                loadPrefectureData(prefCode).then(() => {
                    executeUI();
                });
            } else {
                executeUI();
            }

            function executeUI() {
                requestAnimationFrame(() => {
                    requestAnimationFrame(() => {
                        // 👇 追加: 県が選択された時に確実に道路レイヤーを表示
                        if (!window.isSpotScreen() && typeof showRoadLayers === 'function') showRoadLayers();

                        if (typeof showPrefSpots === 'function') showPrefSpots();
                        if (typeof initAreaUI === 'function') initAreaUI();
                        if (typeof renderPrefWeather === 'function') renderPrefWeather();
                        if (!options.keepRadar && typeof renderCrowdImage === 'function') renderCrowdImage();
                        if (typeof disableAreaSwipe === 'function') disableAreaSwipe();
                        if (typeof enablePrefSwipe === 'function') enablePrefSwipe();
                        
                        if (typeof syncGuideSections === 'function') syncGuideSections();

                        // ★ preloadShop01 は loadPrefectureData のキュー側で実行されるため、ここでは呼ばない（二重登録防止）
                        
                        if (typeof renderSpotRanking === 'function') renderSpotRanking();
                        const waitTasks = [waitTileLayerReady(window.gsiLayer, 1000)];
                        if (window._prefMarkersReadyPromise && typeof window._prefMarkersReadyPromise.then === 'function') {
                            waitTasks.push(window._prefMarkersReadyPromise);
                        }
                        unlockBackNavigationWhenReady({
                            gen: navGen,
                            showButton: true,
                            promises: waitTasks,
                            maxWaitMs: 2200
                        });
                    });
                });
            }
        };

        const center = window.map.getCenter();

        const isSameLoc =
            Math.abs(center.lat - window.prefData.lat) < 0.0001 &&
            Math.abs(center.lng - window.prefData.lng) < 0.0001 &&
            window.map.getZoom() === Number(targetZoom);

        if (drawn || isSameLoc) {
            finalizePref();
        } else {
            window.map.once('moveend', finalizePref);
        }
    }
};

function filterMergedGridSpots(allSpots, map, gridSize = 5) {
    if (!map || !Array.isArray(allSpots) || allSpots.length === 0) return [];

    const occupiedCells = new Set();
    const result = [];

    for (let i = 0; i < allSpots.length; i++) {
        const spot = allSpots[i];
        if (!spot.lat || !spot.lng) continue;

        const p = map.latLngToLayerPoint([spot.lat, spot.lng]);
        const cellX = Math.floor(p.x / gridSize);
        const cellY = Math.floor(p.y / gridSize);
        const cellKey = `${cellX}_${cellY}`;

        if (!occupiedCells.has(cellKey)) {
            occupiedCells.add(cellKey);
            result.push(spot);
        }
    }

    return result;
}
async function loadPrefectureData(prefCode) {
    window.fishData = null; 
    window.currentPref = prefCode;

    const region = window.currentRegion || 'KANTO';
    window.regionUrl = `/${region}/${region}_region.csv`;
    window.fishUrl = `/${region}/${prefCode}_fish.json`;
    window.prefShopUrl = `/${region}/${prefCode}_shop.json`;

    try {
        // ★ 1. 共通ローダーで選択県だけを取得し、全県データの集約を待たずに進む
        // キャッシュ済みの場合も allspotData への反映を保証するため、毎回共通ローダーを通す。
        await loadPrefectureSpots(prefCode, 'select');

        // 2. 座標データ(prefData/areaData/spotData)だけ即座に展開して返す（タイル描画を止めない！）
        await loadLocationJSON();

        // 県データ一式の展開完了後に、気象庁アラートを更新する
        // 県画面では、アラート取得関数の成否に関係なくJNNラベルを表示する

        if (jnnRow && window.currentPref === prefCode) {
            jnnRow.style.display = 'flex';
        }
        if (window.currentPref === prefCode && window.prefData && typeof getAlertText === 'function') {
            getAlertText(window.prefData);
        }
        
        // ★ 3. 天気JSONとショップJSONは、ベースタイルの描画完了キュー(runAfterTileLoad)で取得する
        // 既存の画面遷移・後続処理を維持し、天気取得部分だけ新JSON関数へ切り替える。
        const loadDeferredPrefExtras = async () => {
            if (typeof AddWeatherV2 === 'function') {
                await AddWeatherV2(prefCode, 'select').catch(() => null);
            } else {
                await Promise.all([
                    typeof AddWeather === 'function' ? AddWeather(prefCode, 'select').catch(() => null) : Promise.resolve(null),
                    typeof AddInLandWeather === 'function' ? AddInLandWeather(prefCode, 'select').catch(() => null) : Promise.resolve(null)
                ]);
            }

            if (window.currentPref === prefCode) {
                if (typeof renderPrefWeather === 'function') renderPrefWeather();
                if (window.currentAreaId && typeof phase1menu === 'function') {
                    phase1menu(window.currentAreaId);
                }
                if (window.phase2Initialized && window.map && typeof showNearestSpotName === 'function') {
                    showNearestSpotName(window.map);
                }
                if (window.currentSpotId && window.spotData) {
                    const activeSpot = (window.currentAreaId
                        ? window.spotData.find(s => String(s.areaId) === String(window.currentAreaId) && String(s.individualId) === String(window.currentSpotId))
                        : null)
                        || window.spotData.find(s => String(s.individualId) === String(window.currentSpotId));
                    if (activeSpot && activeSpot.whether && typeof renderSpotWeatherV2 === 'function') {
                        renderSpotWeatherV2(activeSpot.whether);
                    }
                }
            }

            if (window.markerControl && typeof window.markerControl.preloadShop01 === 'function') {
                await window.markerControl.preloadShop01(window.prefShopUrl);
                if (window.currentSpotId && window.spotData && typeof renderShopSection === 'function') {
                    const activeSpot = (window.currentAreaId
                        ? window.spotData.find(s => String(s.areaId) === String(window.currentAreaId) && String(s.individualId) === String(window.currentSpotId))
                        : null)
                        || window.spotData.find(s => String(s.individualId) === String(window.currentSpotId));
                    if (activeSpot) renderShopSection(activeSpot);
                }
            }
        };
        if (typeof window.runAfterTileLoad === 'function') {
            window.runAfterTileLoad(loadDeferredPrefExtras);
        } else {
            loadDeferredPrefExtras();
        }

        return true; 
    } catch (err) {
        console.error("データ読み込みエラー:", err);
        return false;
    }
}
/**
 * sitemap.xmlから記事データを抽出し、カテゴリ別に描画する
 * （スクロール・タッチ連動で初期ロードの2.4MB渋滞を完全にゼロにする版）
 */
function loadAndRenderCategorizedArticles(targetDomId) {
    const targetDom = document.getElementById(targetDomId);
    if (!targetDom) return;

    targetDom.innerHTML = '<div style="text-align:center; padding:20px; color:#666;">記事を読み込み中...</div>';

    let isLoaded = false;

    const executeLoad = async () => {
        if (isLoaded) return;
        isLoaded = true;

        window.removeEventListener('scroll', executeLoad);
        window.removeEventListener('touchstart', executeLoad);
        window.removeEventListener('mousemove', executeLoad);

        try {
            const res = await fetch('sitemap.xml', { priority: 'low' });
            if (!res.ok) throw new Error('サイトマップの取得に失敗しました');
            const xmlText = await res.text();

            const categoryOrder = [];
            const groupedArticles = {};

            // ★ DOMParserによる巨大なXMLツリー構築（Long Task）を廃止し、4msスライスの正規表現で高速抽出する
            const urlBlocks = xmlText.split('<url>');
            let idx = 1;

            await new Promise(resolve => {
                function parseXmlSlice() {
                    const start = performance.now();
                    while (idx < urlBlocks.length && (performance.now() - start) < 4) {
                        const block = urlBlocks[idx++];
                        if (!block.includes('ARTICLE_DATA:')) continue;

                        const locMatch = block.match(/<loc>([^<]+)<\/loc>/);
                        const dataMatch = block.match(/<!--\s*ARTICLE_DATA:\s*(\{[\s\S]*?\})\s*-->/);
                        if (!locMatch || !dataMatch) continue;

                        try {
                            const articleData = JSON.parse(dataMatch[1]);
                            articleData.url = locMatch[1].trim();
                            const cat = articleData.category || "その他の記事";
                            if (!categoryOrder.includes(cat)) {
                                categoryOrder.push(cat);
                                groupedArticles[cat] = [];
                            }
                            groupedArticles[cat].push(articleData);
                        } catch (e) {
                            console.error('記事JSONパースエラー:', e);
                        }
                    }
                    if (idx < urlBlocks.length) {
                        requestAnimationFrame(parseXmlSlice);
                    } else {
                        resolve();
                    }
                }
                requestAnimationFrame(parseXmlSlice);
            });

            let html = '';

            categoryOrder.forEach(categoryName => {
                const articles = groupedArticles[categoryName];
                if (articles.length === 0) return;

                html += `<h2 class="category-heading">${categoryName}</h2>`;

                const compactItems = articles.filter(a => a.style === "compact");
                const listItems = articles.filter(a => a.style !== "compact"); 

                if (compactItems.length > 0) {
                    html += `<div class="article-list">`;
                    compactItems.forEach(item => {
                        html += `
                        <a href="${item.url}" class="article-card">
                            <img src="${item.image}" alt="${item.title}" width="320" height="240" loading="lazy" decoding="async" fetchpriority="low">
                            <h3>${item.title}</h3>
                        </a>`;
                    });
                    html += `</div>`;
                }

                if (listItems.length > 0) {
                    html += `<div class="menu-list">`;
                    listItems.forEach(item => {
                        html += `
                        <a href="${item.url}" class="menu-card">
                            <div class="menu-thumb">
                                <img src="${item.image}" alt="${item.title}" width="60" height="60" loading="lazy" decoding="async" fetchpriority="low">
                            </div>
                            <div class="menu-text">
                                <h3>${item.title}</h3>
                            </div>
                        </a>`;
                    });
                    html += `</div>`;
                }
            });

            if (html === '') {
                html = '<div style="text-align:center; padding:20px; color:#666;">記事が見つかりませんでした。</div>';
            }

            targetDom.innerHTML = html;

        } catch (err) {
            console.error(err);
            targetDom.innerHTML = '<div style="color:red; padding:20px;">記事の読み込みに失敗しました。</div>';
        }
    };

    // ★ IntersectionObserver で記事枠が画面に近づいた瞬間（400px手前）に自動で読み込む（初期表示のLCP乗っ取りを完全防止）
    if ('IntersectionObserver' in window) {
        const observer = new IntersectionObserver((entries) => {
            if (entries.some(entry => entry.isIntersecting)) {
                observer.disconnect();
                executeLoad();
            }
        }, { rootMargin: '400px 0px' });
        observer.observe(targetDom);
    } else {
        executeLoad();
    }
}

// =======================
// STATE
// =======================
let pinchStartDistance = 0;
let pinchTriggered = false;

// =======================
// DISTANCE
// =======================
function getDistance(touches) {
    const dx = touches[0].clientX - touches[1].clientX;
    const dy = touches[0].clientY - touches[1].clientY;
    return Math.sqrt(dx * dx + dy * dy);
}

// =======================
// TOUCH START
// =======================
document.addEventListener('touchstart', (e) => {

    if (e.touches.length !== 2) return;

    // 状態だけ記録（Leafletには触らない）
    pinchStartDistance = getDistance(e.touches);
    pinchTriggered = false;

}, { passive: true });

// =======================
// TOUCH MOVE
// =======================
document.addEventListener('touchmove', (e) => {

    if (e.touches.length !== 2) return;

    const currentDistance = getDistance(e.touches);

    const isPinchIn = currentDistance < pinchStartDistance - 20;

    if (!pinchTriggered && isPinchIn) {

        pinchTriggered = true;

        if (typeof isZoomGuardFailed === 'function' && isZoomGuardFailed()) return;

        //const isPrefView = !window.currentAreaId && !window.currentSpotId;
        //if (isPrefView) return;

        if (typeof goBack === 'function') goBack(window.currentHash);
    }

}, { passive: true });

// =======================
// RESET
// =======================
document.addEventListener('touchend', (e) => {

    if (e.touches.length > 0) return;

    pinchStartDistance = 0;
    pinchTriggered = false;

});



function renderCrowdImage() {

    // ==================================================
    // マップ生成前に呼ばれた場合
    // ==================================================
    if (!window.map) {
        if (!window._crowdRenderRetry) {
            window._crowdRenderRetry = setInterval(() => {
                if (window.map) {
                    clearInterval(window._crowdRenderRetry);
                    window._crowdRenderRetry = null;
                    renderCrowdImage();
                }
            }, 100);
        }
        return;
    }

    // ==================================================
    // 初回だけON
    // ==================================================
    if (typeof window.radarEnabled === 'undefined') {
        window.radarEnabled = true;
    }

    // ==================================================
    // トグルボタン
    // ==================================================
    const btn = document.getElementById('radarToggleBtn');

    if (btn && !btn.dataset.listenerAdded) {
        btn.dataset.listenerAdded = "true";
        btn.addEventListener('click', function() {
            window.radarEnabled = !window.radarEnabled;
            renderCrowdImage();
        });
    }

    // ==================================================
    // OFF時
    // ==================================================
    if (!window.radarEnabled) {
        removeCrowdImage();

        const header = document.querySelector('.menu-header-row');
        const cloudsUI = document.getElementById('cloudsUI');
        const headerText = document.querySelector('.header-text');

        if (header) {
            header.style.display = '';
        }

        if (cloudsUI) {
            cloudsUI.textContent = '';
        }

        if (headerText) {
            if (window.todayTide) {
                headerText.textContent = formatDate(window.todayTide.date) + " " + window.todayTide.tide;
            } else {
                headerText.textContent = "";
            }
        }

        if (btn) {
            btn.style.display = '';
            btn.textContent = '雨雲';
            btn.classList.add('is-off');
            btn.classList.remove('active');
        }
        return;
    }

    // ==================================================
    // ON表示
    // ==================================================
    if (btn) {
        btn.style.display = '';
        btn.textContent = '雨雲';
        btn.classList.remove('is-off');
        btn.classList.add('active');
    }

    // ★ここが重要：以前のレイヤーとタイマーを破棄し、世代を更新する
    removeCrowdImage();
    const myGeneration = window.crowdRenderGeneration;

    // ==================================================
    // Pane準備
    // ==================================================
    if (!window.map.getPane('rainPane')) {
        window.map.createPane('rainPane');
        const pane = window.map.getPane('rainPane');
        pane.style.zIndex = 9999;
        pane.style.pointerEvents = 'none';
    }

    // ==================================================
    // 時刻計算
    // ==================================================
    const now = new Date();
    const nowJST = new Date(
        now.toLocaleString("en-US", { timeZone: "Asia/Tokyo" })
    );

    const hour = nowJST.getHours();
    const slots = [0, 3, 6, 9, 12, 15, 18, 21];

    // ==================================================
    // スライド用データ (KANTO固定)
    // ==================================================
    const images = [];
    let baseIndex = slots.findIndex(h => h > hour);
    let dayOffset = 0;

    if (baseIndex === -1) {
        baseIndex = 0;
        dayOffset = 1;
    }

    for (let i = 0; i < 4; i++) {
        const rawIndex = baseIndex + i;
        const idx = rawIndex % slots.length;
        const extraDay = Math.floor(rawIndex / slots.length);

        const tmpDate = new Date(nowJST);
        tmpDate.setDate(tmpDate.getDate() + dayOffset + extraDay);

        const y = tmpDate.getFullYear();
        const m = String(tmpDate.getMonth() + 1).padStart(2, "0");
        const d = String(tmpDate.getDate()).padStart(2, "0");
        const hh = String(slots[idx]).padStart(2, "0");

        images.push({
            url: `js/crowdsimg/KANTO_${y}-${m}-${d}_${hh}.png`,
            displayHour: slots[idx] + ((dayOffset + extraDay) * 24)
        });
    }

    // ==================================================
    // header UI
    // ==================================================
    const header = document.querySelector('.menu-header-row');
    const cloudsUI = document.getElementById('cloudsUI');
    const headerText = document.querySelector('.header-text');

    if (!header || !cloudsUI || !headerText) {
        return;
    }

    header.style.display = '';

    // ==================================================
    // Phase2では雨雲UIを出さない
    // ==================================================
    if (window.phase2Initialized) {
        cloudsUI.textContent = '';
        if (window.todayTide) {
            headerText.textContent = formatDate(window.todayTide.date) + " " + window.todayTide.tide;
        } else {
            headerText.textContent = "";
        }
        return;
    }

    // ==================================================
    // 初期UI
    // ==================================================
    let currentIndex = 0;
    // ロード中は準備中のような表示にしておくのもありですが、一旦時間を表示しておきます
    cloudsUI.textContent = `雨雲レーダー ${images[0].displayHour}時`;

    if (window.todayTide) {
        headerText.textContent = formatDate(window.todayTide.date) + " " + window.todayTide.tide;
    } else {
        headerText.textContent = "";
    }

    // ==================================================
    // 座標・ズーム固定
    // ==================================================
    const best = {
        lat: 37.25,
        lng: 138.3,
        baseZoom: 7
    };

    function calcBounds() {
        const z = window.map.getZoom();
        const scale = Math.pow(2, z - best.baseZoom);
        const size = 512 * scale;

        const topLeft = window.map.project([best.lat, best.lng], z);
        const bottomRight = topLeft.add([size, size]);

        return L.latLngBounds(
            window.map.unproject(topLeft, z),
            window.map.unproject(bottomRight, z)
        );
    }

    const MAX_OPACITY = 0.6;
    window.crowdLayers = [];
    let loadedCount = 0;

    // ==================================================
    // ★ 4枚のレイヤーを「完全に透明」で事前配置＆ロード待機
    // ==================================================
    images.forEach((img, index) => {
        const layer = L.imageOverlay(
            img.url,
            calcBounds(),
            {
                pane: 'rainPane',
                opacity: 0 // 全て透明でセット
            }
        );

        // CSSトランジションを付与してフワッとした切り替えを有効化
        layer.on('add', () => {
            const el = layer.getElement();
            if (el) el.style.transition = 'opacity 1.2s ease-in-out';
        });

        // ロード完了監視
        layer.on('load', () => {
            loadedCount++;
                // 4枚すべて読み込み終わったら自動発火
            if (loadedCount === images.length && myGeneration === window.crowdRenderGeneration) {
                startCrowdSlideshow();
            }
        });

        // 読み込みエラー時も止まらないようにする
        layer.on('error', () => {
            loadedCount++;
                if (loadedCount === images.length && myGeneration === window.crowdRenderGeneration) {
                startCrowdSlideshow();
            }
        });

        layer.addTo(window.map);
        window.crowdLayers.push(layer);
    });


    // 互換性のため
    window.crowdLayer = window.crowdLayers[0];

    // ==================================================
    // Zoom変更
    // ==================================================
    window.crowdZoomHandler = function() {
        if (myGeneration !== window.crowdRenderGeneration) return;
        window.crowdLayers.forEach(layer => {
            if (layer) layer.setBounds(calcBounds());
        });
    };
    window.map.on("zoomend", window.crowdZoomHandler);

    // ==================================================
    // ★ 準備完了で発火するスライドショー処理
    // ==================================================
    function startCrowdSlideshow() {
        if (myGeneration !== window.crowdRenderGeneration || !window.radarEnabled) {
                return;
        }

        // 1. 最初の一枚目をフワッとフェードイン
        window.crowdLayers[0].setOpacity(MAX_OPACITY);

        // 2. スライドショー開始 (DOMの生成・破棄は一切行わない)
        window.crowdSlideTimer = setInterval(() => {
            if (myGeneration !== window.crowdRenderGeneration || !window.radarEnabled) {
                clearInterval(window.crowdSlideTimer);
                return;
            }

            const oldIndex = currentIndex;
            currentIndex = (currentIndex + 1) % images.length;
            
            cloudsUI.textContent = `雨雲レーダー ${images[currentIndex].displayHour}時`;

            const oldLayer = window.crowdLayers[oldIndex];
            const newLayer = window.crowdLayers[currentIndex];

            // CSSトランジションに任せて opacity を切り替えるだけ
            if (newLayer) newLayer.setOpacity(MAX_OPACITY);
            if (oldLayer) oldLayer.setOpacity(0);

            window.crowdLayer = newLayer;

        }, 2500);
    }
}


function removeCrowdImage() {
    // 描画世代を無効化
    window.crowdRenderGeneration = (window.crowdRenderGeneration || 0) + 1;

    // スライド停止
    if (window.crowdSlideTimer) {
        clearInterval(window.crowdSlideTimer);
        window.crowdSlideTimer = null;
    }

    // フェード処理管理庫のクリア（念のため残す）
    if (window.crowdFadeTimers) {
        window.crowdFadeTimers.forEach(timer => clearInterval(timer));
        window.crowdFadeTimers.clear();
    }

    // Zoomイベント解除
    if (window.map && window.crowdZoomHandler) {
        window.map.off("zoomend", window.crowdZoomHandler);
    }
    window.crowdZoomHandler = null;

    // ==================================================
    // ★ 4枚保持しているレイヤーを一括削除
    // ==================================================
    if (window.map) {
        const layersToRemove = [];

        // 保持している配列から回収
        if (window.crowdLayers && window.crowdLayers.length > 0) {
            window.crowdLayers.forEach(layer => {
                if (layer) layersToRemove.push(layer);
            });
        }
        
        if (window.crowdLayer && !layersToRemove.includes(window.crowdLayer)) {
            layersToRemove.push(window.crowdLayer);
        }

        // rainPane内に残っている ImageOverlay も全回収（無限増殖対策）
        window.map.eachLayer(layer => {
            if (layer instanceof L.ImageOverlay && layer.options && layer.options.pane === 'rainPane') {
                if (!layersToRemove.includes(layer)) {
                    layersToRemove.push(layer);
                }
            }
        });

        // 一括削除
        layersToRemove.forEach(layer => {
            if (window.map.hasLayer(layer)) {
                window.map.removeLayer(layer);
            }
        });
    }

    window.crowdLayer = null;
    window.crowdLayers = [];

    // ==================================================
    // header制御
    // ==================================================
    const header = document.querySelector('.menu-header-row');
    if (!header) return;

    const cloudsUI = document.getElementById('cloudsUI');
    const headerText = document.querySelector('.header-text');

    if (window.phase2Initialized) {
        if (cloudsUI) cloudsUI.textContent = '';
    } else {
        if (cloudsUI) cloudsUI.textContent = '';
        if (headerText) headerText.textContent = '';
    }

    const leftEmpty = !cloudsUI || cloudsUI.textContent === '';
    const rightEmpty = !headerText || headerText.textContent === '';

    if (leftEmpty && rightEmpty) {
        header.style.display = 'none';
    } else {
        header.style.display = '';
    }
}

// ■ 旧天気適用関数の互換ラッパー（実体はJSON）
// ================================
function applyWeatherToSpots(weatherData, targetArray = window.spotData, inlandOnly = false) {
    const weatherMap = weatherData && weatherData.spots;
    if (!weatherMap || typeof weatherMap !== 'object' || !Array.isArray(targetArray)) return;

    for (const spot of targetArray) {
        if (inlandOnly) {
            const spotId = String(spot.individualId || spot.id || "");
            if (!spotId.startsWith("L")) continue;
        }
        const weather = weatherMap[spot.name];
        if (weather) spot.whether = weather;
    }
}

// 旧関数名・引数形式を維持
function applyWhetherToSpots(weatherData, targetArray = window.spotData) {
    applyWeatherToSpots(weatherData, targetArray, false);
}

function applyInlandWhetherToSpots(weatherData, targetArray = window.spotData) {
    applyWeatherToSpots(weatherData, targetArray, true);
}



function getCurrentAreaIndividualId() {
    if (!window.currentAreaId) return null;
    const raw = String(window.currentAreaId).trim();
    if (!raw) return null;

    if (Array.isArray(window.areaData) && window.areaData.length > 0) {
        const matched = window.areaData.find(a =>
            String(a.areaId + '_' + a.individualId) === raw ||
            String(a.individualId) === raw
        );
        if (matched && matched.individualId != null) {
            return String(matched.individualId);
        }
    }

    const idx = raw.lastIndexOf('_');
    return idx !== -1 ? raw.slice(idx + 1) : raw;
}

// ==========================================
// ★ 全画面共通：「特徴と使い方」「スポット詳細」「エリアガイド」の表示同期
//   REGION       : 特徴と使い方のみ表示（エリアガイドはクリア＆非表示）
//   PREF         : エリアガイドのみ表示（全エリア折りたたみ）
//   AREA / Phase2: エリアガイドのみ表示（選択中エリアを開き、選択外を閉じる）
//   SPOT         : スポット詳細のみ表示（特徴と使い方・エリアガイドは非表示）
// ==========================================
function syncGuideSections(options = {}) {
    const siteGuideBox = document.getElementById('siteGuideBox');
    const accessBox = document.getElementById('accessInfoBox');
    const areaGuide = document.getElementById('area-guide');

    const params = new URLSearchParams(window.location.search);
    const hasPref = !!(window.currentPref || params.get('pref'));
    const hasSpot = !!(window.currentSpotId || params.get('spot'));
    const hasArea = !!window.currentAreaId;

    // 1. 「特徴と使い方」：REGION画面のみ表示
    if (siteGuideBox) {
        siteGuideBox.style.display = hasPref ? 'none' : 'block';
    }

    // 2. スポット詳細枠：SPOT画面以外ではクリアして非表示
    if (!hasSpot) {
        if (typeof clearAccessInfo === 'function') {
            clearAccessInfo();
        } else if (accessBox) {
            accessBox.innerHTML = '';
            accessBox.style.display = 'none';
        }
    }

    // 3. 「◯◯県の釣りエリアガイド」
    if (!areaGuide) return;

    // REGION または SPOT 画面では非表示（REGION時は中身もクリア）
    if (!hasPref || hasSpot) {
        areaGuide.style.display = 'none';
        if (!hasPref) {
            areaGuide.innerHTML = '';
            areaGuide.classList.remove('active-mode');
            areaGuide.classList.add('collapsed-mode');
            areaGuide.dataset.renderedPref = '';
        }
        return;
    }

    // PREF / AREA / Phase2 画面
    if (!window.prefData) {
        areaGuide.style.display = 'none';
        return;
    }

    const prefKey = String(window.currentPref || window.prefData.name || '');
    const needsRender =
        areaGuide.dataset.renderedPref !== prefKey ||
        !areaGuide.querySelector('.area') ||
        (!hasArea && areaGuide.classList.contains('active-mode'));

    if (needsRender) {
        renderAreaGuide();
    } else {
        areaGuide.style.display = 'block';
    }

    if (hasArea) {
        const areaIndivId = getCurrentAreaIndividualId();
        if (areaIndivId) {
            openArea(areaIndivId, !!options.scrollToArea);
        }
    } else {
        // PREF画面：全エリアを折りたたみ状態（collapsed-mode）に統一
        areaGuide.classList.remove('active-mode');
        areaGuide.classList.add('collapsed-mode');
        areaGuide.querySelectorAll('.area').forEach(areaEl => {
            areaEl.classList.remove('active');
            const text = areaEl.querySelector('.text');
            if (text) text.classList.remove('open');
        });
    }
}
window.syncGuideSections = syncGuideSections;

function openArea(areaKey, isUserAction = false) {
  const container = document.getElementById('area-guide');
  if (!container) return;

  // SPOT画面ではエリアガイドを非表示のままにする
  if (typeof window.isSpotScreen === 'function' && window.isSpotScreen()) {
    container.style.display = 'none';
    return;
  }

  if (!container.querySelector('.area') && window.prefData) {
    renderAreaGuide();
  }

  container.style.display = 'block';

  // モード切替（選択中エリアのみ開き、選択外を閉じる）
  container.classList.remove('collapsed-mode');
  container.classList.add('active-mode');

  const targetKey = String(areaKey || '');

  document.querySelectorAll('#area-guide .area').forEach(area => {
    const text = area.querySelector('.text');

    if (String(area.id) === targetKey) {
      area.classList.add('active');
      if (text) text.classList.add('open');

      // 先頭寄せ
      if (container.children[1] && container.children[1] !== area) {
        container.insertBefore(area, container.children[1]);
      }

      // ★ ユーザー操作時のみスクロール
      if (isUserAction) {
        area.scrollIntoView({ behavior: 'smooth', block: 'end' });
      }
    } else {
      area.classList.remove('active');
      if (text) text.classList.remove('open');
    }
  });
}

// ================================
// ■ エリアガイドの描画（グローバル関数として復活）
// ================================
function renderAreaGuide() {
    const container = document.getElementById('area-guide');
    if (!container) return; // コンテナがない場合はスキップ

    // REGION画面・SPOT画面では表示しない
    if (!window.prefData || !window.currentPref || (typeof window.isSpotScreen === 'function' && window.isSpotScreen())) {
        container.style.display = 'none';
        return;
    }

    container.style.display = 'block';
    container.dataset.renderedPref = String(window.currentPref || window.prefData.name || '');
    container.innerHTML = `<h2>${window.prefData.notes}の釣りエリアガイド</h2>`;

    if (!window.areaData) return;

    window.areaData.forEach(area => {
        const div = document.createElement('div');
        div.className = 'area';
        div.id = area.individualId; // ←ここ重要

        div.innerHTML = `
            <h3>${area.name}</h3>
            <div class="text">${area.notes}</div>
        `;

        div.querySelector('h3').addEventListener('click', async () => {
            if (typeof selectArea !== 'function' && typeof window.ensureAreaLoader === 'function') {
                await window.ensureAreaLoader();
            }
            
            // =====================================================
            // ★ Phase2（ズーム13固定 ＆ OSMタイル）状態の判定
            // =====================================================
            let isPhase2OSM = false;
            if (window.map && window.map.getZoom() === 13) {
                window.map.eachLayer(layer => {
                    if (layer instanceof L.TileLayer) {
                        const url = layer._url || '';
                        if (url.includes('openstreetmap')) {
                            isPhase2OSM = true;
                        }
                    }
                });
            }

            // =====================================================
            // ★ スポット画面 または Phase2状態から別のエリアへ飛ぶ時のクリーンアップ処理
            // =====================================================
            if (window.currentSpotId || isPhase2OSM) {
                
                // 1. ズーム・移動制限の解除
                if (typeof stopZoomGuard === 'function') stopZoomGuard();
                if (window.map) {
                    window.map.dragging.enable();
                    window.map.scrollWheelZoom.enable();
                    window.map.doubleClickZoom.enable();
                    window.map.touchZoom.enable();
                    window.map.setMinZoom(0);
                    window.map.setMaxZoom(18);
                    window.map.setMaxBounds(null);
                    if (window.map.options) window.map.options.maxBoundsViscosity = 0;

                    // 2. 特殊レイヤー（魚・マーカー等）の消去
                    if (window.fishLayer) window.map.removeLayer(window.fishLayer);
                    if (window.phase2Group) window.phase2Group.clearLayers();
                    if (window.phase1Group) window.phase1Group.clearLayers();
                    if (window.areaSpotLayer) window.areaSpotLayer.clearLayers();

                    // 3. 拡大用タイルの剥離と基本タイル（ort）への切り替え
                    window.map.eachLayer(layer => {
                        if (layer === window.gsiLayer) return;
                        if (layer instanceof L.TileLayer) {
                            const url = layer._url || '';
                            if (url.includes('seamlessphoto') || url.includes('openstreetmap')) {
                                window.map.removeLayer(layer);
                            }
                        }
                    });
                    
                    window._tileFadeGen = (window._tileFadeGen || 0) + 1;
                    if (!window.gsiLayer) {
                        window.gsiLayer = L.tileLayer(window.gsiLayers.ort).addTo(window.map);
                    } else {
                        if (window.gsiLayer._url !== window.gsiLayers.ort) {
                            window.gsiLayer.setUrl(window.gsiLayers.ort);
                        }
                        if (!window.map.hasLayer(window.gsiLayer)) {
                            window.gsiLayer.addTo(window.map);
                        }
                        window.gsiLayer.setZIndex(100);
                        window.gsiLayer.setOpacity(1);
                    }
                    
                    // Phase2の無効化処理があれば呼ぶ
                    if (typeof disablePhase2 === 'function') disablePhase2(window.map);
                }

                // 4. UI（DOM）の完全クリア
                if (typeof clearSpotWeatherV2 === 'function') clearSpotWeatherV2();
                if (typeof toggleTideGraphMode === 'function') toggleTideGraphMode(false);
                if (typeof removeWeekItem === 'function') removeWeekItem();
                if (typeof resetWeatherUI === 'function') resetWeatherUI();
                if (typeof clearSub2Weather === 'function') clearSub2Weather();
                
                // 👇 【ここに追加】エリア移動時にアクセス情報を必ず閉じる
                if (typeof clearAccessInfo === 'function') clearAccessInfo();

                
                const nearest = document.getElementById("nearest-spot");
                if (nearest) nearest.textContent = "";

                // 5. 状態のリセット
                window.currentSpotId = null;
            }

            // =====================================================
            // ★ 新しいエリアへのジャンプ処理（routeUpdateは直打ち専用なので呼ばない）
            // =====================================================
            
            // クエリをセット
            if (window.prefData) setIdealQuery('pref', window.prefData.notes);
            setIdealQuery('area', area.name);
            setIdealQuery('spot', null);

            // システム変数の更新
            window.currentAreaId = area.areaId + '_' + area.individualId;
            window.currentSpotId = null;

            // routeUpdateを完全に排除し、直接selectAreaを呼ぶ
            if (typeof selectArea === 'function') selectArea(area);
            
            syncGuideSections({ scrollToArea: true });
        });

        container.appendChild(div);
    });
}

function resetAreaGuide() {
  const container = document.getElementById('area-guide');
  if (!container) return;

  if (!window.currentPref || !window.prefData) {

  container.classList.remove('active-mode');
  container.classList.add('collapsed-mode');

  // ★ DOMを初期状態（元のエリア順・折りたたみ）に戻す
  renderAreaGuide();
}

// ■ 旧天気CSV関数名の互換ラッパー（実体は県別JSON）

// 既存呼び出しとの互換性のため関数名・引数を維持する。
// URLから県コードを取り出し、JSONを取得する。CSV fetch / parse は行わない。
function loadWhetherCSV(url) {
    const match = url && url.match(/\/([A-Za-z0-9-]+)_whether\.(?:csv|json)(?:$|[?#])/);
    const prefCode = match ? match[1] : window.currentPref;
    if (!prefCode || typeof loadPrefWeatherJson !== 'function') return Promise.resolve(null);
    return loadPrefWeatherJson(prefCode, false);
}

// 旧仕様は地域一括CSVだったが、現在は選択県の内陸JSONを取得する。

function loadInlandWhetherCSV(regionCode) {

    const prefCode = window.currentPref;

}

function preloadCsvFiles(regionCode, prefCode) {
    if (!regionCode || !prefCode) return;

    if (typeof AddWeatherV2 === 'function') {
        AddWeatherV2(prefCode, 'select').catch(() => {});
        return;
    }

    // 旧関数名・引数を暫定維持し、天気JSONの先行取得へ委譲する。
    Promise.all([
        typeof AddWeather === 'function' ? AddWeather(prefCode, 'select') : Promise.resolve(null),
        typeof AddInLandWeather === 'function' ? AddInLandWeather(prefCode, 'select') : Promise.resolve(null)
    ]).catch(() => {});
}

    const prefCode = window.currentPref;

    if (!prefCode) {
        return true;
    }

    try {
let needsBaseLoad = false;

if (!window.prefData) {
    needsBaseLoad = true;

} else if (
    !Array.isArray(window.spotData) ||
    window.spotData.length === 0
) {
    needsBaseLoad = true;

} else {
    const rep = window.spotData.find(s => {
        const typeParts = (s.type || '').split('$');
        return typeParts.includes('representative');
    });

    if (rep && !rep.whether) {
        needsBaseLoad = true;
    }
}

if (needsBaseLoad) {
    await loadPrefectureData(prefCode);
}


        return true;

    } catch (err) {
        console.error("データロードエラー:", err);
        return false;
    }
}

/* ============================================================
 * 釣り行こNAVI 等深線レイヤーコントローラー
 * Leaflet Canvas Renderer管理方式
 * ============================================================ */

const DepthMapController = (function() {
    let mapInstance = null;
    let depthLayers = [];
    let visible = true;
    let loadPromise = null;

    const DEPTH_FILES = [
        '/KANTO/enc_depth20.json',
        '/KANTO/enc_depth50.json',
        '/KANTO/enc_depth100.json',
        '/KANTO/enc_depth150.json',
        '/KANTO/enc_depth200.json'
    ];

    const MULTIPLY_COLORS = {
        20: '#8ca8d9',
        50: '#5a82c4',
        100: '#385799',
        150: '#22326b',
        200: '#191970'
    };

    const EDGE_COLORS = {
        20: '#5a82c4',
        50: '#385799',
        100: '#22326b',
        150: '#191970',
        200: '#0d0d3b'
    };

    let rawDepthData = null;
    let depthDataPromise = null;
    let parsedDepthData = null;
    let parsedDepthPromise = null;

    function fetchDepthJsonSafe(url, retries = 1) {
        return fetch(url, { priority: 'low' }).then(res => {
            if (!res.ok) throw new Error(url + ' HTTP ' + res.status);
            return res.json();
        }).catch(error => {
            if (retries > 0) {
                return new Promise(resolve => setTimeout(resolve, 350))
                    .then(() => fetchDepthJsonSafe(url, retries - 1));
            }
            console.warn('等深線JSON取得スキップ:', url, error && error.message ? error.message : error);
            return null;
        });
    }

    async function loadDepthData() {
        if (rawDepthData) return rawDepthData;
        if (depthDataPromise) return depthDataPromise;

        depthDataPromise = (async () => {
            // 6ファイル一斉発射を避け、2本ずつ段階取得して帯域飽和とタイムアウトを防止する
            const results = new Array(DEPTH_FILES.length).fill(null);
            for (let i = 0; i < DEPTH_FILES.length; i += 2) {
                const batch = [fetchDepthJsonSafe(DEPTH_FILES[i], 2)];
                if (i + 1 < DEPTH_FILES.length) {
                    batch.push(fetchDepthJsonSafe(DEPTH_FILES[i + 1], 2));
                }
                const chunk = await Promise.all(batch);
                results[i] = chunk[0];
                if (i + 1 < DEPTH_FILES.length) results[i + 1] = chunk[1];
            }

            const hasAny = results.some(item => Array.isArray(item) && item.length > 0);
            if (!hasAny) {
                depthDataPromise = null;
                return null;
            }
            const normalized = results.map(item => (Array.isArray(item) ? item : []));
            if (results.every(item => Array.isArray(item))) {
                rawDepthData = normalized;
            } else {
                depthDataPromise = null;
            }
            return normalized;
        })();

        return depthDataPromise;
    }

    function decodePolyline(encoded) {
        const points = [];
        let index = 0;
        let lat = 0;
        let lng = 0;

        while (index < encoded.length) {
            let result = 0;
            let shift = 0;
            let byte;
            do {
                byte = encoded.charCodeAt(index++) - 63;
                result |= (byte & 0x1f) << shift;
                shift += 5;
            } while (byte >= 0x20);
            lat += (result & 1) ? ~(result >> 1) : (result >> 1);

            result = 0;
            shift = 0;
            do {
                byte = encoded.charCodeAt(index++) - 63;
                result |= (byte & 0x1f) << shift;
                shift += 5;
            } while (byte >= 0x20);
            lng += (result & 1) ? ~(result >> 1) : (result >> 1);

            points.push([lat / 1e5, lng / 1e5]);
        }
        return points;
    }

    function loadParsedDepthData() {
        if (parsedDepthData) return Promise.resolve(parsedDepthData);
        if (parsedDepthPromise) return parsedDepthPromise;

        parsedDepthPromise = loadDepthData().then(async sourceData => {
            if (!Array.isArray(sourceData)) {
                parsedDepthPromise = null;
                return null;
            }
            const parsedLayers = [];
            const depths = [20, 50, 100, 150, 200];

            for (let index = 0; index < depths.length; index++) {
                const items = sourceData[index] || [];
                const groupsMap = {};
                let i = 0;

                await new Promise(resolve => {
                    function decodeSlice() {
                        const start = performance.now();

                        while (i < items.length && performance.now() - start < 4) {
                            const item = items[i++];
                            if (!item || !item.p) continue;

                            const points = decodePolyline(item.p);
                            if (points.length >= 3) {
                                const group = item.group || 'default';
                                if (!groupsMap[group]) groupsMap[group] = [];
                                groupsMap[group].push(points.map(point =>
                                    typeof L !== 'undefined' ? L.latLng(point[0], point[1]) : point
                                ));
                            }
                        }

                        if (i < items.length) requestAnimationFrame(decodeSlice);
                        else resolve();
                    }
                    requestAnimationFrame(decodeSlice);
                });

                parsedLayers.push({
                    depth: depths[index],
                    groups: Object.values(groupsMap)
                });
            }

            if (rawDepthData) {
                parsedDepthData = parsedLayers;
            } else {
                parsedDepthPromise = null;
            }
            return parsedLayers;
        }).catch(error => {
            parsedDepthPromise = null;
            throw error;
        });

        return parsedDepthPromise;
    }

    // 等深線の陸地除外マスク。KANTOフォルダのEncoded Polylineを読み込む。
    let MASK_POLYGONS = null;
    let maskDataPromise = null;

    function loadMaskData() {
        if (MASK_POLYGONS) return Promise.resolve(MASK_POLYGONS);
        if (maskDataPromise) return maskDataPromise;

        maskDataPromise = fetchDepthJsonSafe('/KANTO/contourlinesExclusionlist.json', 1)
            .then(items => {
                if (!Array.isArray(items)) {
                    maskDataPromise = null;
                    return [];
                }
                MASK_POLYGONS = items
                    .filter(item => item && item.p)
                    .map(item => decodePolyline(item.p))
                    .filter(points => points.length >= 4);
                return MASK_POLYGONS;
            });

        return maskDataPromise;
    }

    // Leaflet 1.9.4の実際の描画更新メソッド_redrawを拡張する（L未定義時の即時評価エラーを防ぐため遅延生成）。
    let MaskedCanvasRenderer = null;
    function getMaskedCanvasRendererClass() {
        if (MaskedCanvasRenderer) return MaskedCanvasRenderer;
        if (typeof L === 'undefined' || !L.Canvas) return null;
        MaskedCanvasRenderer = L.Canvas.extend({
            _redraw: function() {
                L.Canvas.prototype._redraw.call(this);

                if (!this._map || !this._ctx || !this._bounds) return;

                const ctx = this._ctx;
                ctx.save();
                ctx.globalCompositeOperation = 'destination-out';
                ctx.fillStyle = '#000';

                if (Array.isArray(MASK_POLYGONS) && MASK_POLYGONS.length > 0) MASK_POLYGONS.forEach(polygon => {
                    ctx.beginPath();
                    polygon.forEach((ll, index) => {
                        const point = this._map.latLngToLayerPoint(L.latLng(ll[0], ll[1]));
                        const x = point.x;
                        const y = point.y;
                        if (index === 0) ctx.moveTo(x, y);
                        else ctx.lineTo(x, y);
                    });
                    ctx.closePath();
                    ctx.fill();
                });

                ctx.restore();
            }
        });
        return MaskedCanvasRenderer;
    }

    function createDepthLayers(map, parsedLayers) {
        const RendererClass = getMaskedCanvasRendererClass();
        if (!map || !RendererClass) return;
        parsedLayers.forEach(layerData => {
            const renderer = new RendererClass({
                pane: 'overlayPane',
                padding: 1.5
            });

            const canvas = renderer._container;
            if (canvas) {
                canvas.style.zIndex = '50';
                canvas.style.filter = 'drop-shadow(1px 2px 3px rgba(0, 0, 0, 0.45))';
                canvas.style.mixBlendMode = 'multiply';
                canvas.style.pointerEvents = 'none';
            }

            const groupLayers = [];
            layerData.groups.forEach(rings => {
                const polygon = L.polygon(rings, {
                    renderer: renderer,
                    color: EDGE_COLORS[layerData.depth] || '#191970',
                    weight: 1.5,
                    opacity: 1,
                    lineJoin: 'round',
                    fill: true,
                    fillColor: MULTIPLY_COLORS[layerData.depth] || '#191970',
                    fillOpacity: 0.40,
                    fillRule: 'evenodd',
                    interactive: false,
                    bubblingMouseEvents: false
                });

                polygon.addTo(map);
                groupLayers.push(polygon);
            });

            depthLayers.push({
                depth: layerData.depth,
                renderer: renderer,
                layers: groupLayers
            });
        });
    }

    function removeDepthLayers() {
        depthLayers.forEach(layerData => {
            layerData.layers.forEach(layer => {
                if (mapInstance && mapInstance.hasLayer(layer)) {
                    mapInstance.removeLayer(layer);
                }
            });
            if (mapInstance && mapInstance.hasLayer(layerData.renderer)) {
                mapInstance.removeLayer(layerData.renderer);
            }
        });
        depthLayers = [];
    }

    return {
        loadData: function(options = {}) {
            if (!options.force && typeof window.isSearchBot === 'function' && window.isSearchBot()) {
                window._depthDeferredForBot = true;
                if (typeof bindUserRecoveryListeners === 'function') bindUserRecoveryListeners();
                return Promise.resolve(null);
            }
            return loadDepthData();
        },

        initAndShow: function(leafletMap, options = {}) {
            if (!leafletMap || typeof L === 'undefined') return Promise.resolve();
            mapInstance = leafletMap;

            // 検索ボット時は等深線6本の初期通信をスキップして必須JSONに帯域を譲り、操作時に後から読み込み直す
            if (!options.force && typeof window.isSearchBot === 'function' && window.isSearchBot()) {
                window._depthDeferredForBot = true;
                if (typeof bindUserRecoveryListeners === 'function') bindUserRecoveryListeners();
                return Promise.resolve();
            }

            if (loadPromise) return loadPromise;

            // 先に陸地マスクを1本取得してから等深線本体を段階取得・描画する
            loadPromise = loadMaskData()
                .then(() => loadParsedDepthData())
                .then(parsedLayers => {
                    if (!Array.isArray(parsedLayers) || parsedLayers.length === 0) {
                        loadPromise = null;
                        return [];
                    }
                    createDepthLayers(mapInstance, parsedLayers);
                    return parsedLayers;
                }).catch(error => {
                    loadPromise = null;
                    console.warn('等深線レイヤー生成スキップ:', error && error.message ? error.message : error);
                    return [];
                });

            return loadPromise;
        },

        toggle: function() {
            visible = !visible;
            if (visible && depthLayers.length === 0 && (mapInstance || window.map)) {
                window._depthDeferredForBot = false;
                this.initAndShow(mapInstance || window.map, { force: true });
                return visible;
            }
            depthLayers.forEach(layerData => {
                layerData.layers.forEach(layer => {
                    if (visible) layer.addTo(mapInstance);
                    else if (mapInstance.hasLayer(layer)) mapInstance.removeLayer(layer);
                });
            });
            return visible;
        },

        isActive: function() {
            return visible;
        }
    };
})();
</script>
